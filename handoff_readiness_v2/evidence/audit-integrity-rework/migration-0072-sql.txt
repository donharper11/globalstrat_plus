CREATE OR REPLACE FUNCTION competition_audit_reject_change() RETURNS trigger AS $$
BEGIN
    RAISE EXCEPTION
        'append-only audit table %: % on row % is not permitted',
        TG_TABLE_NAME, TG_OP, OLD.id
        USING ERRCODE = 'insufficient_privilege',
              HINT = 'Audit records are evidence. Correct them by appending.';
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION competition_manifest_reject_change() RETURNS trigger AS $$
BEGIN
    IF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION
            'resolution manifest % cannot be deleted', OLD.id
            USING ERRCODE = 'insufficient_privilege';
    END IF;
    IF OLD.completed_at IS NOT NULL THEN
        RAISE EXCEPTION
            'resolution manifest % completed at %; it is immutable',
            OLD.id, OLD.completed_at
            USING ERRCODE = 'insufficient_privilege',
                  HINT = 'Re-resolving a round writes a new round, not a new manifest.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Kept as its own function so the policy can be interrogated with a database
-- name that is not the caller's, which is the only way to test from inside a
-- test database that a production database would refuse.
CREATE OR REPLACE FUNCTION competition_truncate_is_allowed(db_name text) RETURNS boolean AS $$
BEGIN
    RETURN db_name LIKE 'test\_%'
       AND COALESCE(current_setting('globalstrat.allow_truncate', true), 'off') = 'on';
END;
$$ LANGUAGE plpgsql STABLE;

CREATE OR REPLACE FUNCTION competition_audit_reject_truncate() RETURNS trigger AS $$
BEGIN
    IF competition_truncate_is_allowed(current_database()) THEN
        RETURN NULL;
    END IF;
    RAISE EXCEPTION
        'append-only audit table %: TRUNCATE is not permitted', TG_TABLE_NAME
        USING ERRCODE = 'insufficient_privilege',
              HINT = 'Truncating an audit table destroys the evidence it holds.';
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER competition_decision_audit_event_append_only BEFORE UPDATE OR DELETE ON competition_decision_audit_event FOR EACH ROW EXECUTE FUNCTION competition_audit_reject_change();
