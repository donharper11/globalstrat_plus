# /etc/systemd/system/globalstrat-backend.service
[Unit]
Description=GlobalStrat+ Django Backend (Gunicorn)
After=network.target postgresql.service

[Service]
User=ubuntu
Group=ubuntu
WorkingDirectory=/home/ubuntu/projects/globalstrat+/backend
Environment=GLOBALSTRAT_ENV=production
Environment=PYTHONUNBUFFERED=1
EnvironmentFile=-/etc/globalstrat-plus.env
UMask=0077
ExecStart=/usr/bin/gunicorn -c gunicorn.conf.py globalstrat.wsgi:application
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/globalstrat-narratives.service
[Unit]
Description=GlobalStrat+ durable narrative worker
After=network.target postgresql.service globalstrat-backend.service
Wants=globalstrat-backend.service

[Service]
Type=simple
User=ubuntu
Group=ubuntu
WorkingDirectory=/home/ubuntu/projects/globalstrat+/backend
Environment=GLOBALSTRAT_ENV=production
Environment=PYTHONUNBUFFERED=1
EnvironmentFile=-/etc/globalstrat-plus.env
UMask=0077
ExecStart=/usr/bin/python3 manage.py run_narrative_worker --loop --interval 10
Restart=always
RestartSec=5
# A worker completes its claimed job after SIGTERM.  This exceeds the provider
# call timeout without leaving a shutdown blocked indefinitely.
TimeoutStopSec=180
KillSignal=SIGTERM

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/globalstrat-deadline-scheduler.service
[Unit]
Description=GlobalStrat+ deadline scheduler
After=network-online.target globalstrat-backend.service
Wants=network-online.target

[Service]
Type=oneshot
User=ubuntu
Group=ubuntu
WorkingDirectory=/home/ubuntu/projects/globalstrat+/backend
Environment=GLOBALSTRAT_ENV=production
Environment=PYTHONUNBUFFERED=1
EnvironmentFile=-/etc/globalstrat-plus.env
UMask=0077
TimeoutStartSec=10min
# Deliberately omits --auto-process and --auto-advance: elapsed deadlines
# close the current round only, leaving scoring and advancement to instructors.
ExecStart=/usr/bin/python3 manage.py check_round_deadlines
